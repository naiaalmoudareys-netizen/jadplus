  async function api(path, opts = {}) {
    const headers = { 'content-type': 'application/json', ...(opts.headers || {}) };
    if (token()) headers.authorization = `Bearer ${token()}`;
    let res;
    try {
      res = await fetch(API + path, { credentials: NATIVE ? 'omit' : 'same-origin', ...opts, headers });
    } catch {
      return { status: 0, ok: false, body: { error: 'You appear to be offline. Please check your connection and try again.' } };
    }
    let body = {};
    try { body = await res.json(); } catch { /* empty */ }
    return { status: res.status, ok: res.ok, body };
  }

  async function sfDoc(btn, kind, id) {

        toast(msg); return;
      }
      const name = (/filename="([^"]+)"/.exec(r.headers.get('content-disposition') || '') || [])[1] || `${kind}.pdf`;

      const a = document.createElement('a'); a.href = url; a.download = name; document.body.appendChild(a); a.click(); a.remove();
      setTimeout(() => URL.revokeObjectURL(url), 60000);
    } catch { toast('You appear to be offline. Please try again.'); } finally { btn.disabled = false; btn.innerHTML = label; }
  }

  // ---------- instalment logic ----------
  function instState(i) {
    const bal = n(i.amount) - n(i.paid);
    if (n(i.amount) > 0 && bal <= 1) return { k: 'paid', cls: 'good', icon: 'check', label: 'Paid', bal: 0 };
    if (!i.due) return { k: 'later', cls: 'plain', icon: 'flag', label: /complet|handover/i.test(`${i.name} ${i.desc}`) ? 'On completion' : 'Scheduled', bal };
    const d = daysFromToday(i.due);
    if (d < 0) return { k: 'overdue', cls: 'bad', icon: 'alert', label: `Overdue ${-d} day${d === -1 ? '' : 's'}`, bal };
    if (d <= 30) return { k: 'soon', cls: 'warn', icon: 'clock', label: d === 0 ? 'Due today' : `Due in ${d} day${d === 1 ? '' : 's'}`, bal };
    if (n(i.paid) > 0) return { k: 'part', cls: 'info', icon: 'half', label: 'Part paid', bal };
    return { k: 'upcoming', cls: 'plain', icon: 'dots', label: 'Upcoming', bal };
  }

  function dealSummary(deal) {
    const inst = deal.instalments || [];
    const total = inst.reduce((s, i) => s + n(i.amount), 0) || n(deal.price);
    const paid = inst.reduce((s, i) => s + Math.min(n(i.paid), n(i.amount) || n(i.paid)), 0);
    const states = inst.map((i) => ({ i, s: instState(i) }));
    const overdue = states.filter((x) => x.s.k === 'overdue');
    const overdueSum = overdue.reduce((s, x) => s + x.s.bal, 0);
    const open = states.filter((x) => x.s.k !== 'paid' && x.s.bal > 1)
      .sort((a, b) => (a.i.due ? 0 : 1) - (b.i.due ? 0 : 1) || String(a.i.due).localeCompare(String(b.i.due)) || n(a.i.seq) - n(b.i.seq));
    const next = open.find((x) => x.s.k !== 'overdue') || null;
    return { total, paid, pct: total > 0 ? Math.min(100, (paid / total) * 100) : 0, states, overdue, overdueSum, open, next };
  }

  // Salesforce short names -> owner-facing names; completion "31-03-2028" -> "31 Mar 2028"; unit "205-JAD288-B" -> "205".
  function normalise(b) {
    const fix = (u) => {
      if (!u) return;
      const key = projectKey(u.project);
      u.place = PROJECT_PLACES[key] || '';
      u.image = PROJECT_IMAGES[key] || '';
      u.film = PROJECT_FILMS[key] || null;
      if (u.project) u.project = PROJECT_NAMES[key] || u.project;
      const m = String(u.completion || '').match(/^(\d{1,2})[-/.](\d{1,2})[-/.](\d{4})$/);
      if (m) u.completion = fmtDate(`${m[3]}-${m[2].padStart(2, '0')}-${m[1].padStart(2, '0')}`);
      const raw = u.number || u.name || '';
      u.number = String(raw).split('-')[0] || raw;
      if (/^residential$/i.test(u.type || '')) u.type = !u.bedrooms ? null : /studio/i.test(u.bedrooms) ? 'Studio' : `${u.bedrooms} bedroom`;
      if (u.building) u.building = u.building.replace(/^JAD ?288[- ]?/i, '').replace(/^([A-C])$/, 'Block $1') || u.building;
    };
    (b.deals || []).forEach((d) => fix(d.unit));
    (b.updates || []).forEach((u) => { if (u.project) u.project = PROJECT_NAMES[projectKey(u.project)] || u.project; });
    (b.posts || []).forEach((p) => {
      if (p.project) p.project = PROJECT_NAMES[projectKey(p.project)] || p.project;
      if (p.unit) fix(p.unit);
    });
    b.posts = b.posts || [];
  }

    const idx = data.deals.findIndex((d) => d.id === saved);
    dealIdx = idx >= 0 ? idx : 0;

    showApp();
  }

  // Cash flow: one stacked bar per instalment (paid at the base, still-to-pay above, 2px gap). One axis; tap/hover for details.
  function drawCashflow(box, d) {
    const rows = (d.instalments || []).map((i) => ({ i, s: instState(i) }))
      .sort((a, b) => (a.i.due ? 0 : 1) - (b.i.due ? 0 : 1) || String(a.i.due).localeCompare(String(b.i.due)) || n(a.i.seq) - n(b.i.seq));
    if (!rows.length) { box.innerHTML = '<div class="empty">No instalments yet.</div>'; return; }
    const W = 760, H = 240, padL = 48, padR = 8, padT = 14, padB = 34;
    const max = Math.max(...rows.map((r) => n(r.i.amount)), 1);
    const step = niceStep(max / 4); const top = Math.ceil(max / step) * step;
    const iw = (W - padL - padR) / rows.length; const bw = Math.max(4, Math.min(26, iw - 10));
    const y = (v) => padT + (H - padT - padB) * (1 - v / top);
    const todayIdx = rows.findIndex((r) => r.i.due && r.i.due >= dubaiToday());
    let svg = `<svg viewBox="0 0 ${W} ${H}" role="img" aria-label="Cash flow by instalment">`;
    for (let v = 0; v <= top + 1; v += step) {
      svg += `<line class="grid-line" x1="${padL}" x2="${W - padR}" y1="${y(v)}" y2="${y(v)}"/><text class="axis-label" x="${padL - 10}" y="${y(v) + 4}" text-anchor="end">${short(v)}</text>`;
    }
    const labelEvery = Math.ceil(rows.length / 12);
    rows.forEach((r, k) => {
      const x = padL + k * iw + (iw - bw) / 2;
      const amt = n(r.i.amount); const paid = Math.min(n(r.i.paid), amt); const rest = Math.max(0, amt - paid);
      const restColor = r.s.k === 'overdue' ? COLORS.overdue : COLORS.due;
      const rad = Math.min(4, bw / 2);
      if (rest > 0) svg += roundTop(x, y(amt), bw, Math.max(0, y(paid) - y(amt) - (paid > 0 ? 2 : 0)), rad, restColor);
      if (paid > 0) svg += rest > 0 ? `<rect x="${x}" y="${y(paid)}" width="${bw}" height="${Math.max(0, y(0) - y(paid))}" fill="${COLORS.paid}"/>` : roundTop(x, y(paid), bw, y(0) - y(paid), rad, COLORS.paid);
      svg += `<rect class="bar-hit" data-k="${k}" x="${padL + k * iw}" y="${padT}" width="${iw}" height="${H - padT - padB}"/>`;
      if (k % labelEvery === 0) {
        const lab = r.i.due ? fmtDate(r.i.due).replace(/^\d+ /, '').replace(/ \d{2}(\d{2})$/, " '$1") : 'Completion';
        svg += `<text class="axis-label" x="${x + bw / 2}" y="${H - padB + 18}" text-anchor="middle">${esc(lab)}</text>`;
      }
    });
    if (todayIdx > 0) {
      const tx = padL + todayIdx * iw;
      svg += `<line x1="${tx}" x2="${tx}" y1="${padT}" y2="${H - padB}" stroke="${COLORS.due}" stroke-dasharray="2 4" stroke-width="1.2"/><text class="axis-label" x="${tx + 6}" y="${padT + 10}" style="fill:var(--bronze);font-weight:600;letter-spacing:.12em">TODAY</text>`;
    }
    svg += `<line x1="${padL}" x2="${W - padR}" y1="${y(0)}" y2="${y(0)}" stroke="rgba(57,69,81,.3)"/></svg><div class="tip" id="cf-tip"></div>`;
    box.innerHTML = svg;
    const tip = $('#cf-tip', box);
    $$('.bar-hit', box).forEach((h) => {
      const show = (ev) => {
        const r = rows[Number(h.dataset.k)]; const amt = n(r.i.amount); const paid = Math.min(n(r.i.paid), amt);
        tip.innerHTML = `<b>${esc(r.i.name || '')}</b><br>${r.i.due ? `Due ${esc(fmtDate(r.i.due))}` : esc(r.s.label)}<br>Amount ${money(amt)}<br>Paid ${money(paid)}${amt - paid > 1 ? `<br>${r.s.k === 'overdue' ? 'Overdue' : 'To pay'} ${money(amt - paid)}` : ''}`;
        const rect = box.getBoundingClientRect(); const hr = h.getBoundingClientRect();
        const left = Math.max(0, Math.min(rect.width - tip.offsetWidth, hr.left - rect.left + hr.width / 2 - tip.offsetWidth / 2));
        tip.style.left = `${left}px`; tip.style.top = `${Math.max(0, hr.top - rect.top - tip.offsetHeight + 30)}px`; tip.classList.add('show');
        if (ev && ev.type === 'click') ev.stopPropagation();
      };
      h.addEventListener('mouseenter', show); h.addEventListener('click', show);
      h.addEventListener('mouseleave', () => tip.classList.remove('show'));
    });
    document.addEventListener('click', () => tip.classList.remove('show'));
  }
  function roundTop(x, y0, w, h, r, fill) {
    if (h <= 0) return '';
    r = Math.min(r, h);
    return `<path d="M${x},${y0 + h} L${x},${y0 + r} Q${x},${y0} ${x + r},${y0} L${x + w - r},${y0} Q${x + w},${y0} ${x + w},${y0 + r} L${x + w},${y0 + h} Z" fill="${fill}"/>`;
  }
  function niceStep(raw) { const p = 10 ** Math.floor(Math.log10(raw || 1)); const f = raw / p; return (f <= 1 ? 1 : f <= 2 ? 2 : f <= 2.5 ? 2.5 : f <= 5 ? 5 : 10) * p; }

  // photos are shrunk on the phone (max 1600px JPEG) before upload
  function shrink(file) {
    return new Promise((ok, fail) => {
      const img = new Image(); const url = URL.createObjectURL(file);
      img.onload = () => { const k = Math.min(1, 1600 / Math.max(img.width, img.height)); const c = document.createElement('canvas');
        c.width = Math.round(img.width * k); c.height = Math.round(img.height * k); c.getContext('2d').drawImage(img, 0, 0, c.width, c.height);
        URL.revokeObjectURL(url); c.toBlob((b) => (b ? ok(b) : fail()), 'image/jpeg', 0.82); };
      img.onerror = () => fail(); img.src = url;
    });
  }

  // Photo: cropped to a square and resized to 320px on the device before it is sent.
  function setPhoto(file) {
    if (!file || !/^image\//.test(file.type)) return;
    const img = new Image(); const url = URL.createObjectURL(file);
    img.onload = async () => {
      const s = Math.min(img.width, img.height); const c = document.createElement('canvas'); c.width = c.height = 320;
      c.getContext('2d').drawImage(img, (img.width - s) / 2, (img.height - s) / 2, s, s, 0, 0, 320, 320);
      URL.revokeObjectURL(url);
      if (await saveProfile({ photo: c.toDataURL('image/jpeg', 0.82) })) toast('Photo updated');
    };
    img.onerror = () => toast('That photo could not be read. Please try another.');
    img.src = url;
  }
